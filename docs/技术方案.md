## 一、总体架构

### 1.1 架构原则

- **渐进式改造**：先验证核心体验，再扩展功能，避免一次性重写
- **接口抽象**：embedding、LLM、vector store 通过接口接入，便于切换实现
- **可观测性**：关键链路（检索、生成）保留 trace，方便调试和评估
- **测试驱动**：每个新功能必须有对应测试，CI 自动验证

```javascript
// stores/chat.js
import { defineStore } from "pinia"
import { ref, computed } from "vue"

export const useChatStore = defineStore("chat", () => {
  const messages = ref([])
  const isLoading = ref(false)
  const streamingContent = ref("")
  const sources = ref([])
  const retrievalTrace = ref(null)

  // 计算属性
  const hasMessages = computed(() => messages.value.length > 0)

  // Actions
  function addMessage(message) {
    messages.value.push(message)
  }

  function updateStreamingContent(content) {
    streamingContent.value = content
  }

  function appendStreamingChunk(chunk) {
    streamingContent.value += chunk
  }

  function finishStreaming() {
    if (streamingContent.value) {
      messages.value.push({
        role: "assistant",
        content: streamingContent.value,
        sources: sources.value,
        timestamp: Date.now()
      })
    }
    streamingContent.value = ""
    isLoading.value = false
  }

  function reset() {
    messages.value = []
    streamingContent.value = ""
    sources.value = []
    retrievalTrace.value = null
  }

  return {
    messages, isLoading, streamingContent, sources, retrievalTrace,
    hasMessages, addMessage, updateStreamingContent, appendStreamingChunk,
    finishStreaming, reset
  }
})
```

#### SSE 流式处理

```javascript
// composables/useStreaming.js
import { ref } from "vue"

export function useStreaming() {
  const isStreaming = ref(false)
  const abortController = ref(null)

  async function startStream(url, payload, onChunk, onError) {
    abortController.value = new AbortController()
    isStreaming.value = true

    try {
      const response = await fetch(url, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
        signal: abortController.value.signal
      })

      const reader = response.body.getReader()
      const decoder = new TextDecoder()

      while (true) {
        const { done, value } = await reader.read()
        if (done) break

        const chunk = decoder.decode(value, { stream: true })
        const lines = chunk.split("\n\n")

        for (const line of lines) {
          if (line.startsWith("data: ")) {
            const data = line.slice(6)
            if (data === "[DONE]") {
              isStreaming.value = false
              return
            }
            onChunk(JSON.parse(data))
          }
        }
      }
    } catch (error) {
      if (error.name !== "AbortError") {
        onError?.(error)
      }
    } finally {
      isStreaming.value = false
    }
  }

  function stopStream() {
    abortController.value?.abort()
    isStreaming.value = false
  }

  return { isStreaming, startStream, stopStream }
}
```

#### 文档轮询（优化后）

```javascript
// composables/useDocuments.js
import { ref, computed, onMounted, onUnmounted } from "vue"
import { useKnowledgeBaseStore } from "@/stores/knowledgeBase"

export function useDocuments() {
  const documents = ref([])
  const isLoading = ref(false)
  const pollingTimer = ref(null)
  const pollingFailures = ref(0)

  const POLL_INTERVAL = 1500
  const POLL_MAX_INTERVAL = 10000

  const hasProcessing = computed(() =>
    documents.value.some(doc => doc.status === "processing")
  )

  async function loadDocuments(kbId, notifyOnError = true) {
    if (!kbId) {
      documents.value = []
      return
    }

    isLoading.value = true
    try {
      const { data } = await listDocuments(kbId)
      documents.value = data
      pollingFailures.value = 0
      checkStatusChanges(data)
    } catch (error) {
      pollingFailures.value++
      if (notifyOnError) showError(error, "文档加载失败")
    } finally {
      isLoading.value = false
      schedulePolling()
    }
  }

  function schedulePolling() {
    stopPolling()
    if (!hasProcessing.value) return

    const delay = Math.min(
      POLL_INTERVAL * Math.pow(2, pollingFailures.value),
      POLL_MAX_INTERVAL
    )

    pollingTimer.value = setTimeout(() => {
      loadDocuments(selectedKbId.value, false)
    }, delay)
  }

  function stopPolling() {
    if (pollingTimer.value) {
      clearTimeout(pollingTimer.value)
      pollingTimer.value = null
    }
  }

  function checkStatusChanges(newDocs) {
    newDocs.forEach(doc => {
      const oldDoc = documents.value.find(d => d.id === doc.id)
      if (oldDoc?.status === "processing" && doc.status === "finished") {
        ElMessage.success(`文档「${doc.title}」已完成入库`)
      }
      if (oldDoc?.status === "processing" && doc.status === "failed") {
        ElMessage.error(`文档「${doc.title}」处理失败`)
      }
    })
  }

  onUnmounted(stopPolling)

  return { documents, isLoading, hasProcessing, loadDocuments, stopPolling }
}
```

```python
# services/llm_service.py（改造后）
from abc import ABC, abstractmethod

class LLMService(ABC):
    @abstractmethod
    def chat(
        self,
        system_prompt: str,
        user_prompt: str,
        history: list[dict[str, str]] | None = None,
        model: str | None = None,
        temperature: float = 0.2,
    ) -> dict[str, Any]: ...

    @abstractmethod
    def stream_chat(
        self,
        system_prompt: str,
        user_prompt: str,
        history: list[dict[str, str]] | None = None,
        model: str | None = None,
        temperature: float = 0.2,
    ) -> Generator[str, None, None]: ...

class QwenLLMService(LLMService):
    def stream_chat(self, system_prompt, user_prompt, history=None, model=None, temperature=0.2):
        if not self.client:
            raise HTTPException(status_code=500, detail="LLM not configured")

        safe_history = [
            msg for msg in (history or [])
            if msg.get("role") in {"user", "assistant"} and msg.get("content")
        ]

        response = self.client.chat.completions.create(
            model=model or self.model,
            messages=[
                {"role": "system", "content": system_prompt},
                *safe_history,
                {"role": "user", "content": user_prompt},
            ],
            temperature=temperature,
            stream=True,
        )

        for chunk in response:
            if chunk.choices[0].delta.content:
                yield chunk.choices[0].delta.content

@lru_cache(maxsize=1)
def get_llm_service() -> LLMService:
    return QwenLLMService()
```

### 3.2 SSE 流式接口

```python
# api/chat.py（新增流式端点）
from fastapi import APIRouter, Depends
from fastapi.responses import StreamingResponse
from app.services.rag_service import RagService, get_rag_service
from app.services.llm_service import get_llm_service

router = APIRouter(prefix="/api", tags=["chat"])

@router.post("/chat/stream")
async def chat_stream(
    payload: ChatRequest,
    db: Session = Depends(get_db),
    rag_service: RagService = Depends(get_rag_service),
):
    knowledge_base = db.get(KnowledgeBase, payload.kb_id)
    if knowledge_base is None:
        raise HTTPException(status_code=404, detail="Knowledge base not found")

    config = typed_settings(db)

    # 检索阶段
    retrieval_query = rag_service.build_retrieval_query(payload.question, [])
    sources = vector_store_service.search(
        kb_id=payload.kb_id,
        query=retrieval_query,
        top_k=int(config["top_k"]),
    )

    if not sources:
        yield f"data: {json.dumps({"type": "no_evidence", "content": NO_EVIDENCE_ANSWER})}\n\n"
        yield "data: [DONE]\n\n"
        return

    # 构建上下文
    context = rag_service.build_context(sources)
    user_prompt = f"参考资料：\n{context}\n\n用户问题：{payload.question}"

    # 流式生成
    llm_service = get_llm_service()
    full_content = ""

    async def generate():
        nonlocal full_content
        for chunk in llm_service.stream_chat(
            SYSTEM_PROMPT,
            user_prompt,
            model=str(config["qwen_model"]),
            temperature=float(config["temperature"]),
        ):
            full_content += chunk
            yield f"data: {json.dumps({"type": "content", "content": chunk}, ensure_ascii=False)}\n\n"

        yield f"data: {json.dumps({"type": "sources", "sources": sources}, ensure_ascii=False)}\n\n"
        yield "data: [DONE]\n\n"

    return StreamingResponse(generate(), media_type="text/event-stream")
```

### 4.1 多轮查询重构

```python
# services/rag_service.py
class RagService:
    def build_retrieval_query(
        self,
        question: str,
        history: list[dict[str, str]],
        max_turns: int = 3,
        max_chars_per_turn: int = 200,
    ) -> str:
        user_turns = [
            msg["content"]
            for msg in history
            if msg.get("role") == "user" and msg.get("content")
        ][-max_turns:]

        if not user_turns:
            return question

        context = " | ".join(
            turn[-max_chars_per_turn:]
            for turn in user_turns
        )
        return f"历史问题：{context}\n当前问题：{question}"
```

### 4.2 混合检索（向量 + 关键词）

```python
# services/vector_store_service.py
class VectorStoreService:
    def hybrid_search(
        self,
        kb_id: int,
        query: str,
        top_k: int = 5,
        vector_weight: float = 0.7,
        keyword_weight: float = 0.3,
    ) -> list[dict[str, Any]]:
        # 向量检索（返回 top_k * 2 用于融合）
        vector_results = self.search(kb_id, query, top_k=top_k * 2)

        # 关键词检索
        keyword_results = self._keyword_search(kb_id, query, top_k=top_k * 2)

        # 归一化并融合分数
        scored = {}

        for rank, result in enumerate(vector_results):
            chunk_id = result["chunk_id"]
            scored[chunk_id] = {
                **result,
                "final_score": vector_weight * (1 - rank / len(vector_results))
            }

        for rank, result in enumerate(keyword_results):
            chunk_id = result["chunk_id"]
            keyword_score = keyword_weight * (1 - rank / len(keyword_results))
            if chunk_id in scored:
                scored[chunk_id]["final_score"] += keyword_score
            else:
                scored[chunk_id] = {**result, "final_score": keyword_score}

        merged = sorted(scored.values(), key=lambda x: x["final_score"], reverse=True)
        return merged[:top_k]
```

### 4.3 检索评估模块

```python
# services/evaluation_service.py
class EvaluationService:
    def __init__(self):
        self.eval_dataset: list[dict] = []

    def add_eval_case(self, question: str, relevant_chunk_ids: list[int], kb_id: int):
        self.eval_dataset.append({
            "question": question,
            "relevant_chunk_ids": relevant_chunk_ids,
            "kb_id": kb_id,
        })

    def evaluate_recall_at_k(self, k: int = 5) -> dict[str, float]:
        if not self.eval_dataset:
            return {"recall_at_k": 0.0, "coverage": 0.0}

        hits = 0
        total = len(self.eval_dataset)

        for case in self.eval_dataset:
            results = vector_store_service.search(
                kb_id=case["kb_id"],
                query=case["question"],
                top_k=k,
            )
            retrieved_ids = {r["chunk_id"] for r in results}
            if retrieved_ids & set(case["relevant_chunk_ids"]):
                hits += 1

        return {
            "recall_at_k": hits / total,
            "coverage": len([c for c in self.eval_dataset if c.get("relevant_chunk_ids")]) / total,
            "total_cases": total,
        }

    def evaluate_mrr(self) -> float:
        if not self.eval_dataset:
            return 0.0

        reciprocal_ranks = []
        for case in self.eval_dataset:
            results = vector_store_service.search(
                kb_id=case["kb_id"],
                query=case["question"],
                top_k=10,
            )
            for rank, result in enumerate(results, 1):
                if result["chunk_id"] in case["relevant_chunk_ids"]:
                    reciprocal_ranks.append(1.0 / rank)
                    break
            else:
                reciprocal_ranks.append(0.0)

        return sum(reciprocal_ranks) / len(reciprocal_ranks)
```

```python
# core/auth.py
from fastapi import Depends, HTTPException, Security
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import JWTError, jwt
from datetime import datetime, timedelta

security = HTTPBearer()

class AuthService:
    def __init__(self):
        self.secret_key = settings.jwt_secret_key
        self.algorithm = "HS256"
        self.access_token_expire_minutes = 60 * 24  # 24 小时

    def create_access_token(self, user_id: int) -> str:
        expire = datetime.utcnow() + timedelta(minutes=self.access_token_expire_minutes)
        payload = {"sub": str(user_id), "exp": expire}
        return jwt.encode(payload, self.secret_key, algorithm=self.algorithm)

    async def get_current_user(
        self,
        credentials: HTTPAuthorizationCredentials = Security(security),
    ) -> int:
        token = credentials.credentials
        try:
            payload = jwt.decode(token, self.secret_key, algorithms=[self.algorithm])
            user_id = int(payload["sub"])
            return user_id
        except JWTError:
            raise HTTPException(status_code=401, detail="Invalid token")

auth_service = AuthService()

async def get_current_user_id(
    credentials: HTTPAuthorizationCredentials = Security(security),
) -> int:
    return await auth_service.get_current_user(credentials)
```

```python
    def test_no_evidence_returns_default(self, rag_service):
        result = rag_service.answer(
            kb_id=999,
            question="完全不相关的问题",
        )
        assert result["answer"] == NO_EVIDENCE_ANSWER

    def test_multi_turn_retrieval(self, rag_service, sample_chunks):
        rag_service.index_chunks(sample_chunks)

        history = [
            {"role": "user", "content": "Spark RDD 是什么？"},
            {"role": "assistant", "content": "RDD 是..."},
        ]

        result = rag_service.answer(
            kb_id=1,
            question="它有什么特点？",
            history=history,
        )
        assert any("Spark" in s["content"] for s in result["sources"])
```
